WITH base AS (
    SELECT
        voucher_id,
        voucher_line_num,
        ROW_NUMBER() OVER (
            PARTITION BY voucher_id, voucher_line_num
            ORDER BY
                CASE field_name
                    WHEN 'DEPTID' THEN 1
                    WHEN 'FUND' THEN 2
                  WHEN 'PROJECT_ACTIVITY' THEN 3
                END
        ) AS row_num,
        value
    FROM (
        SELECT
            v.voucher_id,
            vl.voucher_line_num,
            d.deptid,
            d.fund_code,

            /* Project + Activity logic based on your rules */
          CASE 
                WHEN TRIM(d.project_id) <> ' ' AND TRIM(d.activity_id) <>  ' '
                    THEN d.project_id || ' ' || d.activity_id       -- both exist

                WHEN TRIM(d.project_id) <> ' ' AND (d.activity_id IS NULL OR TRIM(d.activity_id) = '')
                    THEN d.project_id                               -- only project exists

                WHEN (d.project_id IS NULL OR TRIM(d.project_id) = '') 
                     AND TRIM(d.activity_id) <> ' '
                    THEN d.activity_id                              -- only activity exists

                ELSE NULL                                           -- both missing ? suppress
            END AS project_activity

        FROM ps_voucher v
        JOIN ps_voucher_line vl 
          ON vl.business_unit = v.business_unit
         AND vl.voucher_id = v.voucher_id
        JOIN ps_distrib_line d 
          ON vl.business_unit = d.business_unit
         AND vl.voucher_id = d.voucher_id
         AND vl.voucher_line_num = d.voucher_line_num
        WHERE v.entry_status <> 'X'
          AND v.close_status <> 'C' 
   ) 
    UNPIVOT (
        value FOR field_name IN (
            deptid AS 'DEPTID',
            fund_code AS 'FUND',
            project_activity AS 'PROJECT_ACTIVITY'
        )
    )
    WHERE TRIM(value) IS NOT NULL or   TRIM(value) <>' ' -- final suppression
)
SELECT
    b.voucher_id AS "*No.",
    b.voucher_line_num AS "*Invoice Line Replacement Data Line No",
    b.row_num AS "*Invoice Line Worktags Line No",
    b.value AS "Worktags"
FROM base b
ORDER BY voucher_id, voucher_line_num, row_num
