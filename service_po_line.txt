WITH shipto_ed AS (
    SELECT
        st.setid,
        st.shipto_id,
        st.descr
    FROM
        ps_shipto_tbl st
    WHERE
        st.eff_status = 'A'
        AND st.effdt = (
            SELECT
                MAX(st2.effdt)
            FROM
                ps_shipto_tbl st2
            WHERE
                st2.setid = st.setid
                AND st2.shipto_id = st.shipto_id
                AND st2.effdt <= SYSDATE
        )
), base AS (
    SELECT
        h.business_unit,
        h.po_id,
        l.line_nbr,
    s.sched_nbr,
       /* CASE
            WHEN substr(TRIM(h.po_id), 1, 3) = 'PO-' THEN TRIM(h.po_id)
            WHEN translate(TRIM(h.po_id), '0123456789', '') IS NULL THEN
                'PO-'
                || lpad(TRIM(h.po_id), 8, '0')
            ELSE
                'PO-' || TRIM(h.po_id)
        END AS po_no*/
        h.po_id as po_no,
        l.inv_item_id,
        l.itm_id_vndr,
        l.unit_of_measure,
        l.category_id,
        l.cntrct_id,
        l.cntrct_line_nbr,
        nvl(nullif(TRIM(l.descr254_mixed2), ''), l.descr254_mixed) AS item_descr,
        s.qty_po,
        s.price_po,
        nvl(s.merchandise_amt,(s.qty_po * s.price_po)) AS extended_amt,
        s.due_dt,
        s.shipto_setid,
        s.shipto_id,
        d.business_unit_gl,
        d.location,
        d.resource_category,
        d.distrib_ln_status,
        h.prepaid_po_flg,
       -- cnt.cntrct_id,
        cnt.CNTRCT_STYLE,
        d.deptid
    FROM
        ps_po_hdr            h
        JOIN ps_po_line           l ON l.business_unit = h.business_unit
                             AND l.po_id = h.po_id
        LEFT OUTER JOIN PS_CNTRCT_HDR cnt ON cnt.setid='SHARE'
                                    and cnt.cntrct_id=l.cntrct_id
        
        JOIN ps_po_line_ship      s ON s.business_unit = l.business_unit
                                  AND s.po_id = l.po_id
                                  AND s.line_nbr = l.line_nbr
                                  AND s.sched_nbr = 1
        JOIN ps_po_line_distrib   d ON d.business_unit = s.business_unit
                                     AND d.po_id = s.po_id
                                     AND d.line_nbr = s.line_nbr
                                     AND d.sched_nbr = s.sched_nbr
                                     AND d.dst_acct_type = 'DST'
                                     AND d.distrib_line_num = 1
    WHERE
        1 = 1
        AND h.po_status NOT IN (
            'C',
            'X'
        )
        AND l.cancel_status <> 'X'
        AND s.cancel_status <> 'X'
        AND d.distrib_ln_status <> 'X'
)
select 
  b.po_no                                                     AS "*No.",
    b.line_nbr                                                  AS "*Service Line Replacement Data Line No",
    b.inv_item_id                                               AS "Item",
    b.po_no    || '-'    || TO_CHAR(b.line_nbr)                 AS "Service Order Line ID",
    b.line_nbr                                                   AS "Line Number",
    b.business_unit_gl                                           AS "Line Company",
    b.itm_id_vndr                                                as "Description",
     CASE
        WHEN TRIM(b.cntrct_id) IS NOT NULL THEN
            TRIM(b.cntrct_id)
            || '-'
            || TO_CHAR(b.cntrct_line_nbr)
        ELSE
            ' '
    END                                                          AS "Supplier Contract Line",
 
  ' ' as "Commodity Code",
  ' ' as "Payment Status",
 ' ' as "Invoice Status",
 ' ' as "Receiving Status",
  b.resource_category   AS "*Resource Category",
 ' '  as "Tax Applicability",
 ' '  as "Tax Code",
 ' '  as "Tax Rate 1",
 ' '  as "Tax Recoverability 1",
' '   as "Tax Option 1",
 ' '  as "Tax Recoverability 2",
' '   as "Tax Option 2",
 ' '  as "Tax Recoverability 3",
' '   as "Tax Option 3",
 ' '  as "Tax Recoverability 4",
 ' '  as "Tax Option 4",
' '   as "Tax Recoverability 5",
' '   as "Tax Option 5",
' '   as "Tax Recoverability 6",
' '   as "Tax Option 6",
b.extended_amt  as "Extended Amount",
TO_CHAR(b.due_dt, 'YYYY-MM-DD')  as "Due Date",
 ' ' as "Start Date",
 ' ' as "End Date",
 ' ' as "Prepaid",
' '  as "Down Payment",
' '  as "Retention",
' '  as "Budget Date",
' '  as "Memo",
 CASE
        WHEN st.descr IS NOT NULL THEN
            b.shipto_id
            || ' - '
            || st.descr
        ELSE
            b.shipto_id
    END as "Ship To Address",
 ' ' as "Ship To Contact Is Employee",
 ' ' as "Ship To Contact Worker ID",
 ' ' as "Requester Is Employee",
 ' ' as "Requester Worker ID",
 ' ' as "Deliver To Location",
 TRIM(b.cntrct_id) as "Supplier Contract",
 ' ' as "Storage Location",
 ' ' as "Close Status"
 FROM
    base        b
    LEFT JOIN shipto_ed   st ON st.setid = b.shipto_setid
                              AND st.shipto_id = b.shipto_id
ORDER BY
    b.po_no,
    b.line_nbr
